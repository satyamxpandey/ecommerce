function calculatePrice(selectedKarat) { let metalCost = 0; const isSilver = ITEM.mainCategory.includes('Silver'); if (isSilver) { const factor = (selectedKarat === '925') ? 0.9250 : 1.0000; metalCost = ITEM.netMetalWt * (STORE_DATA.rates.silver * factor); } else { // Direct ratio relative to 22K benchmark const ratioFrom22k = { "24K": 24 / 22, "22K": 1.0, "18K": 18 / 22, "14K": 14 / 22, "9K": 9 / 22 }; const factor = ratioFrom22k[selectedKarat] || 1.0; metalCost = ITEM.netMetalWt * (STORE_DATA.rates.gold22k * factor); } const subtotal = metalCost + ITEM.makingCharges + ITEM.stoneCost; const gst = subtotal * (STORE_DATA.rates.gst / 100); return { metalCost: Math.round(metalCost), makingCharges: ITEM.makingCharges, stoneCost: ITEM.stoneCost, subtotal: Math.round(subtotal), gst: Math.round(gst), finalPrice: Math.round(subtotal + gst) }; }
